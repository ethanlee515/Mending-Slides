\begin{frame}{Recap: The LMSS patch construction}
  \begin{center}
    \begin{minipage}{0.78\textwidth}
      \begin{algorithm}[H]
        \caption{Noise-flooded decryption}
        \begin{algorithmic}
          \Procedure{$\operatorname{Dec}'$}{$k,\hat{x}$}
            \State $y\gets\operatorname{Dec}_{k}(\hat{x})$
            \State $e'\gets\mathcal{DG}^{n}_{\sigma}$
            \State \Return $y+e'$
          \EndProcedure
        \end{algorithmic}
      \end{algorithm}
    \end{minipage}
  \end{center}
\end{frame}

\begin{frame}{Recap: The LMSS patch intuition}
  \begin{align*}
    \operatorname{Dec}'_{k}(\hat{x})
      &= \operatorname{Dec}_{k}(\hat{x})+e' \\
      &= x+e(k)+e' \\
      &\approx_{\mathrm{dist}} x+e'.
  \end{align*}
  \vspace{0.5em}
  \takeaway{Wide noise hides a small key-dependent shift.}
\end{frame}
