\begin{frame}{The LMSS patch}
  \begin{center}
    \begin{minipage}{0.78\textwidth}
      \begin{algorithm}[H]
        \caption{Noise-flooded decryption}
        \begin{algorithmic}
          \Procedure{$\operatorname{Dec}'_{k}$}{$\hat{y}$}
            \State $y\gets\operatorname{Dec}_{k}(\hat{y})$
            \State $e'\gets\mathcal{DG}^{n}_{\sigma}$
            \State \Return $y+e'$
          \EndProcedure
        \end{algorithmic}
      \end{algorithm}
    \end{minipage}
  \end{center}
\end{frame}

\begin{frame}{LMSS intuition}
  \begin{align*}
    \operatorname{Dec}'_{k}(\hat{y})
      &= \operatorname{Dec}_{k}(\hat{y})+e' \\
      \uncover<2->{&= y+e'} \\
      \uncover<3->{&= f(x)+e(k)+e'} \\
      \uncover<4->{&\approx f(x)+e'.}
  \end{align*}
\end{frame}
