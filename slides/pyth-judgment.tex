\begin{frame}{Our Pythagorean judgment}
  \[
    \vDash_{\mathsf{Pyth}}\{\varphi\}\;
      c_L\approx_{\vec{s}} c_R\;\{\psi\}
  \]
  \uncover<2->{%
    For every initial pair $(\rho_L,\rho_R)$ satisfying $\varphi$,
    there exist $\mathcal P=\{P_i\}_{i=1}^{n}$ and
    $\mathcal Q=\{Q_i\}_{i=1}^{n}$ such that:\par
    \vspace{0.4em}
    {\centering
      $\displaystyle
        P_n=[\![c_L]\!](\rho_L),\qquad
        Q_n=[\![c_R]\!](\rho_R).$\par
    }%
  }%
  \uncover<3->{%
    For every coordinate $i$ and prefix $a$:
    \[
      D_{\mathrm{KL}}(P_i\mid a\;\|\;Q_i\mid a)\leq s_i.
    \]
  }
  \uncover<4->{%
    $\psi$ holds on every possible output of either program.\par
  }
\end{frame}
