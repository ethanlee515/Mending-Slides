\begin{frame}{Our Pythagorean judgment}
  \[
    \vDash_{\mathsf{Pyth}}\{\mathsf{Pre}\}\;
      c_L\approx_s c_R\;\{\mathsf{Post}\}
  \]
  {\small $s=(s_1,\ldots,s_n)$.\par}
  \uncover<2->{%
    For every initial pair $(\rho_L,\rho_R)$ satisfying $\mathsf{Pre}$,
    there exist $\mathcal P=\{P_i\}_{i=1}^{n}$ and
    $\mathcal Q=\{Q_i\}_{i=1}^{n}$ such that:\par
  }
  \uncover<3->{%
    \[
      P_n=[\![c_L]\!](\rho_L),\qquad
      Q_n=[\![c_R]\!](\rho_R).
    \]
  }
  \uncover<4->{%
    For every coordinate $i$ and prefix $a$:
    \[
      D_{\mathrm{KL}}(P_i\mid a\;\|\;Q_i\mid a)\leq s_i.
    \]
  }
  \uncover<5->{%
    $\mathsf{Post}$ holds on every possible output of either program.\par
  }
\end{frame}
