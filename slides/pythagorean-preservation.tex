  \begin{lemma}[Pythagorean preservation]
    Let $\mathcal P,\mathcal Q\in\mathsf{distr}(X^n)$,
    and let $s_1,\ldots,s_n\geq0$.
    For every coordinate $i$ and prefix $a\in X^{i-1}$, suppose the
    conditional KL divergence is finite and
    \[
      D_{\mathrm{KL}}(P_i\mid a\;\|\;Q_i\mid a)
        \leq s_i.
    \]
    Then the final-coordinate marginals satisfy
    \[
      \|P_n-Q_n\|_1
        \leq\sqrt{2\sum_{i=1}^{n}s_i}.
    \]
  \end{lemma}
