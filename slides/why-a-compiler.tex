\begin{frame}{Putting the rules to work}
  If the adversary came in $q$ rounds\ldots
  \begin{align*}
    A^{\mathsf{ODec}} &= \bigl\{
      A_0;\ \mathsf{ODec};\ A_1;\ \cdots;\ \mathsf{ODec};\ A_q\bigr\}
      \\[0.5em]
    A^{\mathsf{ODec}'} &= \bigl\{
      A_0;\ \mathsf{ODec}';\ A_1;\ \cdots;\ \mathsf{ODec}';\ A_q\bigr\}
  \end{align*}
  \medskip
  \pause
  We have
  \[
    \vDash_{\mathsf{Pyth}}\{I^{=}\}\;
      A^{\mathsf{ODec}}\approx_{\vec{s}} A^{\mathsf{ODec}'}\;\{I\}.
  \]
  \begin{center}
    {\small where $\vec{s}=(0,\varepsilon,0,\ldots,\varepsilon,0)$.}
  \end{center}
  \vspace{0.4em}
  \uncover<3->{%
    {\setbeamercolor{takeaway}{fg=MendingInk,bg=MendingCoral!12}
      \takeaway{Whoops: $A$ is \texttt{raw\_code}, with no extra structure.}}
  }
\end{frame}
