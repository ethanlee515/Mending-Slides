\begin{frame}{Main result, revisited}
  \relax
  {\small Assume an IND-CPA-secure, approximately correct FHE scheme $S$.\par}
  \medskip
  {\small For every adaptive $q$-query adversary $\mathcal A$, we construct
  a reduction $\mathcal B_{\mathcal A,q}$ and prove:\par}
  \vspace{0.25em}
  \[
    \Pr\!\left[
      \mathsf{IND\text{-}CPAD}_{\mathsf{NF}_{\gamma}[S],q}(\mathcal A)=1
    \right]
    \leq
    \underbrace{\beta_{\mathsf{CPA}}(\mathcal B_{\mathcal A,q})}_{\text{base FHE}}
    +\underbrace{\frac{\sqrt{qn}}{2\gamma}}_{\text{flooding loss}}.
  \]
  \pause
  \vspace{0.65em}
  \takeaway{Program logic and compiler correctness\\formally verified in Rocq/SSProve.}
\end{frame}
