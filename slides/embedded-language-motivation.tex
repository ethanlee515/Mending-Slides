\begin{frame}[fragile,t]{Why \emph{program} logic}
  \vspace{0.8em}
  \begin{columns}[T,onlytextwidth]
    \column{0.43\textwidth}
      Consider this program:\par
      \begin{lstlisting}[language=Python,basicstyle=\ttfamily\small,
        keywordstyle=\color{MendingTeal}\bfseries,columns=fullflexible,
        keepspaces=true,showstringspaces=false]
a = b = -1

def roll():
    global a, b
    a = randint(1, 6)
    b = randint(1, 6)
    r = a + b
    return r
      \end{lstlisting}
    \column{0.53\textwidth}
      Can we write this as a pure\par
      Rocq / Lean function?\par
      \pause
      \vspace{0.7em}
      \textbf{No.}\par
      \begin{itemize}
        \item Overwriting global variables
        \item Random values
      \end{itemize}
      \vspace{0.8em}
      \pause
      \takeaway{Solution: an \emph{embedded} language.}
  \end{columns}
\end{frame}
