\begin{frame}{Program logic motivations}
  \textbf{Goal:} \texttt{roll()} always returns $r>0$.\par
  \pause
  \vspace{0.6em}
  \begin{center}
    \small
    \renewcommand{\arraystretch}{1.2}
    \begin{tabular}{@{}l@{\hspace{1.2cm}}l@{}}
      \textbf{Code} & \textbf{What we know} \\
      Before the call & \uncover<2->{Nothing} \\
      \texttt{a = randint(1, 6)} & \uncover<3->{\textcolor{MendingTeal}{$a>0$}} \\
      \texttt{b = randint(1, 6)} & \uncover<4->{\textcolor{MendingTeal}{$a>0\;\land\;b>0$}} \\
      \texttt{r = a + b} & \uncover<5->{\textcolor{MendingTeal}{$a>0\;\land\;b>0\;\land\;r=a+b$}} \\
      & \uncover<6->{\textcolor{MendingTeal}{$r>0$} \quad{\scriptsize (consequence)}} \\
      \texttt{return r} & \uncover<7->{\textcolor{MendingTeal}{$r>0$}} \\
    \end{tabular}
  \end{center}
\end{frame}
