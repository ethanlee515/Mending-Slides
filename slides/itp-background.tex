\begin{theorem}[infinitude of primes]
    For all $n$, there exists prime $p$ such that $p\geq n$.
\end{theorem}
\pause
\begin{proof}
    Suppose for contradiction that every prime is less than $n$.
    Let $x = n!+1$, and let $p$ be a prime that divides it.
    By assumption, we have $p < n$.
    We then have $p \mid n!$.
    Therefore $p \mid 1$. Contradiction.
\end{proof}
\end{frame}
