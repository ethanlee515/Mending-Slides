\begin{theorem}[infinitude of primes]
    For every $n\in\mathbb N$, there is a prime $p>n$.
\end{theorem}
\pause
\begin{proof}
    Let $x=n!+1>1$, and choose a prime divisor $p\mid x$.
    Suppose for contradiction that $p\leq n$.
    Then $p\mid n!$, so $p\mid(x-n!)=1$.
    But a prime cannot divide $1$. Therefore $p>n$.
\end{proof}
\end{frame}
