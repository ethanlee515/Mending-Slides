\begin{frame}{Preconditions and postconditions}
  A judgment is a claim about a program.
  \[
    \vDash_{\mathsf{Hoare}}\{\varphi\}\;c\;\{\psi\}
  \]
  \begin{center}
    If $\varphi$ holds before, $\psi$ holds after.
  \end{center}
  \pause
  \vspace{1em}
  \[
    \vDash_{\mathsf{Hoare}}\{\mathsf{true}\}\;
      \texttt{roll()}\;\{r>0\}
  \]
  \begin{center}
    No initial assumptions; every possible return is positive.
  \end{center}
\end{frame}
