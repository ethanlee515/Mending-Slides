\begin{frame}{Relational judgments: additive error}
  \[
    \vDash_{\mathsf{AE}}\{\mathsf{Pre}\}\;
      c_L\approx_{\varepsilon}c_R\;\{\mathsf{Post}\}
  \]
  \uncover<2->{%
    For every initial pair $(\rho_L,\rho_R)$ satisfying $\mathsf{Pre}$:\par
    \vspace{0.4em}
    {\centering
      $\displaystyle
        \exists\pi\in\mathsf{Coupl}\bigl(
          [\![c_L]\!](\rho_L),[\![c_R]\!](\rho_R)\bigr)$\par
      \vspace{0.5em}
      $\displaystyle
        \Pr_{(o_L,o_R)\sim\pi}
          [\mathsf{Post}(o_L,o_R)]\geq1-\varepsilon.$\par
    }%
  }%
  \uncover<3->{%
    \vspace{0.5em}
    \takeaway{Equality postcondition $o_L=o_R$:
      $\bigl\|[\![c_L]\!](\rho_L)-[\![c_R]\!](\rho_R)\bigr\|_1\leq2\varepsilon$.}
  }
\end{frame}
