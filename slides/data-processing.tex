  \begin{theorem}[Data processing inequality]
    Let $X,Y,Z$ be sets, and let
    \[
      D,D' : X\longrightarrow\mathsf{distr}(Y),
      \qquad P : Y\longrightarrow\mathsf{distr}(Z).
    \]
    For every $x\in X$, we have
    \[
      \bigl\|(P\circ D)(x)-(P\circ D')(x)\bigr\|_1
        \leq \bigl\|D(x)-D'(x)\bigr\|_1.
    \]
  \end{theorem}
